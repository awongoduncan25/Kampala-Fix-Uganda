const PILOT = [0.3136, 32.5811];
let pickMap, bigMap, bigLayer, pin=null, picked=null;

function initPickMap(){
  pickMap = L.map("map",{zoomControl:true}).setView(PILOT,14);
  L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png",{maxZoom:19,attribution:"© OpenStreetMap"}).addTo(pickMap);
  pickMap.on("click", e=>setPin(e.latlng));
  setTimeout(()=>pickMap.invalidateSize(),200);
}
function setPin(ll){
  picked = ll;
  const icon = L.divIcon({html:`<div style="background:${COLORS.new};width:18px;height:18px;border-radius:50%;border:3px solid #fff;box-shadow:0 2px 8px rgba(0,0,0,.3)"></div>`,className:"",iconSize:[18,18],iconAnchor:[9,9]});
  if(pin) pin.setLatLng(ll); else pin = L.marker(ll,{icon}).addTo(pickMap);
  document.getElementById("loc").textContent = ll.lat.toFixed(5)+", "+ll.lng.toFixed(5);
}
function initBigMap(){
  if(bigMap) return;
  bigMap = L.map("bigmap").setView(PILOT,14);
  L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png",{maxZoom:19,attribution:"© OpenStreetMap"}).addTo(bigMap);
  bigLayer = L.layerGroup().addTo(bigMap);
}
function refreshBigMap(){
  if(!bigMap) return;
  bigLayer.clearLayers();
  DB.all().forEach(r=>{
    if(!r.lat) return;
    const c = COLORS[r.status];
    const icon = L.divIcon({html:`<div style="background:${c};width:20px;height:20px;border-radius:50%;border:3px solid #fff;box-shadow:0 2px 8px rgba(0,0,0,.3)"></div>`,className:"",iconSize:[20,20],iconAnchor:[10,10]});
    L.marker([r.lat,r.lng],{icon})
      .bindPopup(`<b>${esc(r.title)}</b><br><span class="pill ${r.status}">${LABEL[r.status]}</span><br><small>${r.ref}</small>`)
      .addTo(bigLayer);
  });
}

function fileToDataURL(file){ return new Promise(res=>{ const fr=new FileReader(); fr.onload=()=>res(fr.result); fr.readAsDataURL(file); }); }
function makeRef(){ return "KF-"+Math.random().toString(16).slice(2,7).toUpperCase(); }
function esc(s){ return (s||"").replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

document.getElementById("gps").onclick=()=>{
  if(!navigator.geolocation) return toast("GPS not available on this device","err");
  toast("Getting your location…");
  navigator.geolocation.getCurrentPosition(
    p=>{ setPin({lat:p.coords.latitude,lng:p.coords.longitude}); pickMap.setView([p.coords.latitude,p.coords.longitude],16); toast("Location pinned","ok"); },
    ()=>toast("Could not get GPS. Tap the map instead.","err")
  );
};

document.getElementById("form").onsubmit = async e=>{
  e.preventDefault();
  if(!picked) return toast("Please pin the location on the map.","err");
  const f = document.getElementById("photo").files[0];
  if(!f) return toast("Please add a photo.","err");
  if(f.size > 4*1024*1024) return toast("Photo too large (max 4MB).","err");
  const img = await fileToDataURL(f);
  const r = {
    ref: makeRef(),
    name: document.getElementById("name").value.trim(),
    phone: document.getElementById("phone").value.trim(),
    title: document.getElementById("title").value.trim(),
    desc: document.getElementById("desc").value.trim(),
    cat: document.getElementById("cat").value,
    lat: picked.lat, lng: picked.lng,
    img, status:"new", created: Date.now(),
    ackAt:null, resolvedAt:null, notes:[]
  };
  DB.add(r);
  e.target.reset();
  document.getElementById("preview").style.display="none";
  picked=null; if(pin){pickMap.removeLayer(pin);pin=null;}
  document.getElementById("loc").textContent="No location yet";
  toast("Report submitted — ref "+r.ref,"ok");
  renderList(); renderStats();
  setTimeout(()=>{ document.getElementById("trackRef").value = r.ref; switchTab("track"); document.getElementById("trackBtn").click(); }, 600);
};

function timeAgo(ts){
  const d=(Date.now()-ts)/1000;
  if(d<3600) return Math.round(d/60)+"m ago";
  if(d<86400) return Math.round(d/3600)+"h ago";
  return Math.round(d/86400)+"d ago";
}
function emptyState(msg){
  return `<div class="empty">
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5"><path d="M4 4h16v16H4z"/><path d="M9 9h6M9 13h6M9 17h4"/></svg>
    <p>${msg}</p></div>`;
}
function renderList(){
  const out = document.getElementById("list");
  const a = DB.all();
  if(!a.length){ out.innerHTML = emptyState("No reports yet. Be the first to report a hotspot."); return; }
  out.innerHTML = a.map(r=>`
    <div class="report">
      <img src="${r.img}" alt=""/>
      <div class="body">
        <div class="title">${esc(r.title)}</div>
        <div class="meta">${r.ref} · ${timeAgo(r.created)} <span class="pill ${r.status}">${LABEL[r.status]}</span></div>
      </div>
    </div>`).join("");
}

    <div class="body">
      <div class="title">${esc(r.title)}</div>
      <div class="meta">${r.ref} <span class="pill ${r.status}">${LABEL[r.status]}</span></div>
      <div class="meta" style="margin-top:6px">Reported ${new Date(r.created).toLocaleString()}</div>

document.getElementById("unlock").onclick=()=>{
  if(document.getElementById("pass").value!==CONFIG.passcode) return toast("Wrong passcode","err");
  const p = document.getElementById("respPanel");
  p.classList.remove("hidden");
  const a = DB.all();
  p.innerHTML = a.length ? a.map(r=>`
    <div class="report"><div class="body">
      <div class="title">${esc(r.title)} <small class="meta">${r.ref}</small></div>
      <div class="meta" style="margin-bottom:8px">Status: <span class="pill ${r.status}">${LABEL[r.status]}</span></div>
      <label>Update status</label>
      <select onchange="updateStatus('${r.ref}',this.value)">
        ${STATUS.map(s=>`<option value="${s}" ${s===r.status?"selected":""}>${LABEL[s]}</option>`).join("")}
      </select>
      <label>Note <span class="hint">(optional)</span></label>
      <input onchange="addNote('${r.ref}',this.value)" placeholder="e.g. Truck dispatched, cleared 3pm"/>
    </div></div>`).join("") : emptyState("No reports to action yet.");
  toast("Responder panel unlocked","ok");
};
window.updateStatus=(ref,status)=>{
  const patch={status};
  if(status==="assigned" && !DB.get(ref).ackAt) patch.ackAt=Date.now();
  if(status==="resolved") patch.resolvedAt=Date.now();
  DB.update(ref,patch); renderStats(); renderList();
  toast("Status updated to "+LABEL[status],"ok");
};
window.addNote=(ref,text)=>{
  if(!text.trim()) return;
  const r = DB.get(ref);
  r.notes.push({text, at:Date.now()});
  DB.update(ref,{notes:r.notes});
  toast("Note saved","ok");
};

function renderStats(){
  const a = DB.all();
  const res = a.filter(r=>r.status==="resolved").length;
  document.getElementById("sTotal").textContent=a.length;
  document.getElementById("sRes").textContent=res;
  document.getElementById("sPend").textContent=a.length-res;
}
